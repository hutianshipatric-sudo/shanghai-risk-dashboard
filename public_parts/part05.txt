function alertCards(ind,kind,extra={}){
 let arr=[],dailyStd=ind.vol!=null?ind.vol/Math.sqrt(252):null;
 if(ind.ret1!=null&&dailyStd!=null&&Math.abs(ind.ret1)>Math.max(0.015,1.8*dailyStd))arr.push({level:'red',name:'单日异常波动',desc:`最新单日变化 ${fpct(ind.ret1)}，超过近期波动率对应的异常阈值。短期价格风险正在放大。`});
 if(ind.breakout==='up')arr.push({level:'yellow',name:'向上突破预警',desc:'价格突破近20个交易日高位。若同时成交量/持仓上升，趋势可信度更高；若量能不足，需防范假突破。'});
 if(ind.breakout==='down')arr.push({level:'red',name:'向下突破预警',desc:'价格跌破近20个交易日低位。对采购方有利于成本，但若库存/基本面同步恶化，可能意味着趋势性下行。'});
 if(ind.volPct!=null&&ind.volPct>=0.8)arr.push({level:'red',name:'高波动环境',desc:`20日年化波动率约 ${fpct(ind.vol)}，处于近一年较高分位（约 ${fpct(ind.volPct,0)}）。建议扩大报价安全垫并提高套保关注度。`});
 if(ind.z20!=null&&Math.abs(ind.z20)>=2)arr.push({level:'yellow',name:'价格偏离均值较大',desc:`价格相对20日均线的Z-score约 ${ind.z20.toFixed(2)}。这既可能代表强趋势，也可能意味着短期过度延伸，需结合持仓判断。`});
 if(ind.month&&ind.month.avg&&ind.p){let dev=ind.p/ind.month.avg-1;if(Math.abs(dev)>=0.02)arr.push({level:'yellow',name:'现价明显偏离本月已实现均价',desc:`现价较本月累计均价 ${dev>=0?'高':'低'} ${fpct(Math.abs(dev))}。合同按月均价结算，所以重点是剩余月份能否把均价继续向当前价格方向拉动。`})}
 if(extra.termSpread!=null&&Math.abs(extra.termSpreadPct||0)>=0.003)arr.push({level:'blue',name:'期限结构出现明显价差',desc:`近月与次近月价差约 ${fnum(extra.termSpread,0)}（${fpct(extra.termSpreadPct)}）。近月更贵通常代表近端偏紧，远月更贵则更多体现库存/持有成本或供应宽松。`});
 if(!arr.length)arr.push({level:'green',name:'暂无显著技术性预警',desc:'当前价格、波动和量仓指标未触发主要阈值。仍需关注月均价路径和基本面数据变化。'});
 return `<div class="alerts">${arr.map(a=>`<div class="alert ${a.level}"><div class="top"><div class="name"><span class="dot ${a.level==='red'?'red':a.level==='yellow'?'yellow':a.level==='green'?'green':'blue'}"></span>${a.name}</div><span class="badge ${a.level}">${a.level==='red'?'高关注':a.level==='yellow'?'关注':a.level==='green'?'正常':'结构'}</span></div><div class="desc">${a.desc}</div></div>`).join('')}</div>`;
}
