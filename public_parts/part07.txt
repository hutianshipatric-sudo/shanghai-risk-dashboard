function sinaVarName(symbol){return '__SINA_'+symbol.replace(/[^A-Za-z0-9_]/g,'_')+'_'+Date.now()+'_'+Math.floor(Math.random()*1e6)}
function loadSinaDaily(symbol){
 return new Promise((resolve,reject)=>{
  const vn=sinaVarName(symbol);
  const now=new Date();
  const type=`${now.getFullYear()}_${String(now.getMonth()+1).padStart(2,'0')}_${String(now.getDate()).padStart(2,'0')}`;
  const sc=document.createElement('script');
  const timer=setTimeout(()=>{try{sc.remove()}catch(e){};reject(new Error(symbol+' 请求超时'))},12000);
  sc.src=`https://stock2.finance.sina.com.cn/futures/api/jsonp.php/var%20${vn}=/InnerFuturesNewService.getDailyKLine?symbol=${encodeURIComponent(symbol)}&type=${encodeURIComponent(type)}`;
  sc.async=true;
  sc.onload=()=>{
   clearTimeout(timer);
   try{
    const raw=window[vn];
    delete window[vn]; sc.remove();
    if(!Array.isArray(raw)||!raw.length) throw new Error(symbol+' 返回空数据');
    const rows=raw.map(a=>({date:String(a[0]),open:+a[1],high:+a[2],low:+a[3],close:+a[4],volume:+a[5],open_interest:+a[6],hold:+a[6],settle:(Number.isFinite(+a[7])&&(+a[7])>0)?+a[7]:null}))
      .filter(r=>r.date&&Number.isFinite(r.close)).sort((a,b)=>String(a.date).localeCompare(String(b.date)));
    resolve(rows);
   }catch(e){reject(e)}
  };
  sc.onerror=()=>{clearTimeout(timer);try{sc.remove()}catch(e){};reject(new Error(symbol+' 网络请求失败'))};
  document.head.appendChild(sc);
 });
}

 const d=new Date(); d.setMonth(d.getMonth()+offset);
 return prefix+String(d.getFullYear()).slice(-2)+String(d.getMonth()+1).padStart(2,'0');
}

function mergeCurrentContracts(base,c0,c1,c2){
 const by={}; (base||[]).forEach(r=>by[r.date]={...r});
 const m0=ymSymbol('CU',0),m1=ymSymbol('CU',1),m2=ymSymbol('CU',2);
 const map1=Object.fromEntries((c1||[]).map(r=>[r.date,r]));
 const map2=Object.fromEntries((c2||[]).map(r=>[r.date,r]));
 (c0||[]).forEach(r=>{
  const s1=map1[r.date],s2=map2[r.date];
  by[r.date]={date:r.date,front_symbol:m0,front_settle:r.settle,front_close:r.close,front_volume:r.volume,front_open_interest:r.open_interest,
   second_front_symbol:s1?m1:(s2?m2:null),second_front_settle:s1?s1.settle:(s2?s2.settle:null),second_front_close:s1?s1.close:(s2?s2.close:null),second_front_volume:s1?s1.volume:(s2?s2.volume:null),second_front_open_interest:s1?s1.open_interest:(s2?s2.open_interest:null),
   calendar_symbol:m0,calendar_settle:r.settle,calendar_close:r.close,calendar_volume:r.volume,calendar_open_interest:r.open_interest};
  if(validPrice(by[r.date].front_settle)&&validPrice(by[r.date].second_front_settle)){
   by[r.date].front_second_spread=+by[r.date].front_settle-(+by[r.date].second_front_settle);
   by[r.date].front_second_spread_pct=by[r.date].front_second_spread/(+by[r.date].second_front_settle);
  }
 });
 // If current month contract stopped trading, use next month as front for later dates.
 (c1||[]).forEach(r=>{
  const month=String(r.date).slice(0,7),cur=new Date().toISOString().slice(0,7);
  if(month!==cur)return;
  const existing=by[r.date];
  if(!existing||existing.front_symbol==='CU0'){
   const s2=map2[r.date];
   by[r.date]={date:r.date,front_symbol:m1,front_settle:r.settle,front_close:r.close,front_volume:r.volume,front_open_interest:r.open_interest,
    second_front_symbol:s2?m2:null,second_front_settle:s2?s2.settle:null,second_front_close:s2?s2.close:null,second_front_volume:s2?s2.volume:null,second_front_open_interest:s2?s2.open_interest:null,
    calendar_symbol:null,calendar_settle:null,calendar_close:null,calendar_volume:null,calendar_open_interest:null};
   if(s2&&validPrice(r.settle)&&validPrice(s2.settle)){by[r.date].front_second_spread=+r.settle-(+s2.settle);by[r.date].front_second_spread_pct=by[r.date].front_second_spread/(+s2.settle)}
  }
 });
 return Object.values(by).sort((a,b)=>String(a.date).localeCompare(String(b.date)));
}
